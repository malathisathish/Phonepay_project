🔍 Advanced Analytics & SQL Queries

Complex Data Analysis and Business Intelligence

📊 Key Analytical Queries

-- Top States by Transaction Growth Rate WITH yearly_growth AS ( SELECT "State", "Year", SUM("Transaction_amount") AS total_amount, LAG(SUM("Transaction_amount")) OVER (PARTITION BY "State" ORDER BY "Year") AS prev_year FROM agg_trans_table GROUP BY "State", "Year" ) SELECT "State", ROUND(((total_amount - prev_year) / prev_year * 100), 2) AS growth_rate FROM yearly_growth WHERE prev_year IS NOT NULL ORDER BY growth_rate DESC LIMIT 10;

📈 Transaction Analysis Queries

-- Seasonal Transaction Pattern SELECT "Quarter", AVG("Transaction_count") as avg_count, AVG("Transaction_amount") as avg_amount FROM agg_trans_table GROUP BY "Quarter" ORDER BY "Quarter"; -- Device Brand Performance SELECT "Device_Brand", SUM("Device_Count") as total_users, RANK() OVER (ORDER BY SUM("Device_Count") DESC) as brand_rank FROM agg_user_table GROUP BY "Device_Brand";

🎯 Business Intelligence Queries

-- Insurance Penetration Analysis SELECT d.state_name, ROUND(AVG(ins.total_count::float / users.registered_users * 100), 2) as penetration_rate FROM map_ins_table ins JOIN map_user_table users USING (district, year, quarter) GROUP BY d.state_name ORDER BY penetration_rate DESC; -- High-Value Low-Volume Analysis SELECT "State", "Transaction_type", AVG("Transaction_amount"/"Transaction_count") as avg_transaction_value FROM agg_trans_table WHERE "Transaction_count" < 1000000 GROUP BY "State", "Transaction_type" ORDER BY avg_transaction_value DESC;
